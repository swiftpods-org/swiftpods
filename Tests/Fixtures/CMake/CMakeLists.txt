cmake_minimum_required(VERSION 3.24)
project(SwiftPodsCMake LANGUAGES C CXX OBJC OBJCXX)

add_executable(Demo MACOSX_BUNDLE ../Sources/main.mm)
target_compile_features(Demo PRIVATE cxx_std_17)
target_compile_definitions(Demo PRIVATE CMAKE_FIXTURE=1)
set_target_properties(Demo PROPERTIES
    MACOSX_BUNDLE_GUI_IDENTIFIER "org.swiftpods.cmake-fixture"
    XCODE_GENERATE_SCHEME YES
    XCODE_ATTRIBUTE_CLANG_ENABLE_MODULES YES
    XCODE_ATTRIBUTE_OTHER_CPLUSPLUSFLAGS "$(inherited)"
    XCODE_ATTRIBUTE_OTHER_LDFLAGS "$(inherited)"
    XCODE_ATTRIBUTE_CONFIGURATION_BUILD_DIR "$(BUILD_DIR)/$(CONFIGURATION)$(EFFECTIVE_PLATFORM_NAME)"
    XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER "org.swiftpods.cmake-fixture"
    XCODE_ATTRIBUTE_GCC_TREAT_WARNINGS_AS_ERRORS YES
    XCODE_ATTRIBUTE_SWIFT_TREAT_WARNINGS_AS_ERRORS YES)
